Konomi Documentation

Basic Syntax

Variable Declaration
let name = "value"
let number = 42
let result = 10 + 5 * 2
AI Interaction
ask "What is the meaning of life?"
let question = "How does photosynthesis work?"
ask question
Control Flow
if (age > 18) {
    ask "What career advice would you give?"
} else {
    ask "What subjects should I focus on in school?"
}
Error Handling
try {
    ask "Complex question about quantum physics"
} catch {
    ask "Can you explain that in simpler terms?"
}
Arithmetic Operations
let a = 10
let b = 5
let sum = a + b
let product = a * b
let complex = (a + b) * 2
Comparison Operations
if (value == 10) {
    ask "Equal to 10"
}
if (value > 5) {
    ask "Greater than 5"
}
if (value < 20) {
    ask "Less than 20"
}

Examples

Basic Conversation
let greeting = "Hello, AI!"
ask greeting

let age = 25
if (age >= 18) {
    ask "What are some good career options?"
} else {
    ask "What subjects should I study?"
}
Error Handling Example
try {
    let response = ask "Explain quantum entanglement"
    if (response.length > 100) {
        ask "Can you summarize that?"
    }
} catch {
    ask "Let's try a simpler question"
}
Math Operations
let price = 100
let discount = 20
let final_price = price - (price * discount / 100)
if (final_price < 90) {
    ask "This is a good deal!"
}
